Connect this web app with your Google Apps Script Web App Endpoint.

Kaura Monitoring 3-Sheet Architecture:

  1. Sheet 1 ("Data Tanaman"): Master crop registry (Column A: No, Column B: Nama Tanaman, Column C: Lokasi, Column D: Harga/kg).
  2. Sheet 2 ("Log_Monitoring"): Weekly growth stage logs (Tanggal, No, Nama Tanaman, Lokasi, Perkecambahan, Pertumbuhan Awal, Pertumbuhan Maksimal, Pematangan, Siap Panen, Telah Dipanen, Catatan, Auditor).
  3. Sheet 3 ("Log_Panen"): Independent harvest yield entries (Tanggal, No, Nama Tanaman, Lokasi, Massa Panen (kg), Catatan, Petugas Panen).

Connecting this URL enables dual-sheet reading and multi-sheet writing to Log_Monitoring and Log_Panen.

Updated Google Apps Script Code (3-Sheet Webhook Handler): 

// Apps Script Code for 3 Sheets: "Data Tanaman" (Cell D: Harga/kg), "Log_Monitoring", & "Log_Panen"
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Fetch Master Crop List (Col A: No, Col B: Nama, Col C: Lokasi, Col D: Harga/kg)
    var sheetMaster = ss.getSheetByName("Data Tanaman");
    var dataTanaman = sheetMaster ? sheetMaster.getDataRange().getValues() : [];
    
    // 2. Fetch Log Monitoring Records
    var sheetLog = ss.getSheetByName("Log_Monitoring");
    var logMonitoring = sheetLog ? sheetLog.getDataRange().getValues() : [];
    
    // 3. Fetch Harvest Yield Logs
    var sheetPanen = ss.getSheetByName("Log_Panen");
    var logPanen = sheetPanen ? sheetPanen.getDataRange().getValues() : [];

    return ContentService.createTextOutput(JSON.stringify({ 
      result: "success", 
      dataTanaman: dataTanaman,
      logMonitoring: logMonitoring,
      logPanen: logPanen
    })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);

    // CASE A: Dedicated Harvest Mass Entry (Log_Panen)
    if (data.type === 'yield') {
      var sheetPanen = ss.getSheetByName("Log_Panen");
      if (!sheetPanen) {
        sheetPanen = ss.insertSheet("Log_Panen");
        sheetPanen.appendRow(["Tanggal", "No", "Nama Tanaman", "Lokasi", "Massa Panen (kg)", "Catatan", "Petugas Panen"]);
      }
      sheetPanen.appendRow([
        data.date || new Date().toISOString().split('T')[0],
        data.no,
        data.name,
        data.location,
        data.weight || 0,
        data.notes || "-",
        data.harvester || "Kaura Field Team"
      ]);
      return ContentService.createTextOutput(JSON.stringify({ result: "success" })).setMimeType(ContentService.MimeType.JSON);
    }

    // CASE B: Weekly Growth Stage Inspection (Log_Monitoring)
    var sheetLog = ss.getSheetByName("Log_Monitoring");
    if (!sheetLog) {
      sheetLog = ss.insertSheet("Log_Monitoring");
      sheetLog.appendRow(["Tanggal", "No", "Nama Tanaman", "Lokasi", "Perkecambahan", "Pertumbuhan Awal", "Pertumbuhan Maksimal", "Pematangan", "Siap Panen", "Telah Dipanen", "Catatan", "Auditor"]);
    }
    
    var timestamp = data.date || new Date().toISOString().split('T')[0];
    var inspector = data.inspector || "Kaura Field Team";
    
    if (data.items && data.items.length > 0) {
      data.items.forEach(function(item) {
        sheetLog.appendRow([
          timestamp,
          item.no,
          item.name,
          item.location,
          item.perkecambahan ? "YES" : "NO",
          item.pertumbuhanAwal ? "YES" : "NO",
          item.pertumbuhanMaks ? "YES" : "NO",
          item.pematangan ? "YES" : "NO",
          item.siapPanen ? "READY HARVEST" : "NO",
          item.telahDipanen ? "HARVESTED" : "NO",
          item.notes || "-",
          inspector
        ]);
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
